import numpy as np


# ─────────────────────────────────────────────────────────────────────────────
# PART 1: NAIVE GEMM
# ─────────────────────────────────────────────────────────────────────────────

def gemm_naive(A: np.ndarray, B: np.ndarray) -> np.ndarray:
    """
    Compute C = A @ B using three explicit loops (no np.dot / np.matmul).

    Args:
        A: shape (M, K)
        B: shape (K, N)

    Returns:
        C: shape (M, N)
    """
    # TODO: implement
    
    # M,N,K
    M, K = A.shape
    N = B.shape[1]
    C = np.zeros((M, N), dtype=A.dtype)
    
    for i in range (M):
        for j in range(N):
            for k in range(K):
                C[i][j] += A[i][k]*B[k][j]
    
    return C        


# ─────────────────────────────────────────────────────────────────────────────
# PART 2: TILED GEMM
# ─────────────────────────────────────────────────────────────────────────────

def gemm_tiled(A: np.ndarray, B: np.ndarray, tile_size: int) -> np.ndarray:
    """
    Compute C = A @ B using a tiled (blocked) algorithm.

    The idea: instead of iterating element-by-element, process the matrices
    in (tile_size x tile_size) sub-blocks. For each output tile C[i,j],
    accumulate partial products from corresponding tiles of A and B:

        C_tile[i,j] += A_tile[i,z] @ B_tile[z,j]    for z in 0..K_tiles-1

    You may use np.dot / @ for the *inner* tile-level multiply (size b x b).
    The outer three loops over tile indices must be explicit.

    Assume M, K, N are all divisible by tile_size.

    Args:
        A:         shape (M, K)
        B:         shape (K, N)
        tile_size: block dimension b

    Returns:
        C: shape (M, N)
    """
    # TODO: implement
    
    # M N K
    M, K = A.shape
    N = B.shape[1]
    C = np.zeros((M, N), dtype=A.dtype)
    
    # 按tile
    for i in range(M // tile_size):
        for j in range(N // tile_size):
            for k in range(K // tile_size):
                A_tile = A[i*tile_size:(i+1)*tile_size, k*tile_size:(k+1)*tile_size]
                B_tile = B[k*tile_size:(k+1)*tile_size, j*tile_size:(j+1)*tile_size]
                C_tile = A_tile @ B_tile
                C[i*tile_size:(i+1)*tile_size, j*tile_size:(j+1)*tile_size] += C_tile       
    
    return C
    # pass
