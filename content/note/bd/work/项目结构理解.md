## ES与VikingDB的区别

| **特性**   | **Elasticsearch (ES)**                    | **VikingDB (字节跳动)**                       |
| -------- | ----------------------------------------- | ----------------------------------------- |
| **本质**   | 全文检索搜索引擎                                  | 原生向量数据库 (Vector Database)                 |
| **背景**   | 诞生于 2010 年，基于 Lucene，最初是为了处理文字搜索。         | 字节跳动火山引擎出品，支撑抖音、今日头条等海量推荐业务。              |
| **向量能力** | 后来者。通过 `dense_vector` 类型和 HNSW 算法实现的向量检索。 | 原生者。架构设计之初就围绕向量存储和超大规模相似性检索。              |
| **优势场景** | **混合搜索**。如果你既要搜关键词，又要搜语义（向量），ES 的集成度最高。   | **超大规模、高性能**。处理亿级、十亿级甚至更高维度的向量，延迟更低、吞吐更高。 |

~~~python fold file:plan.py->schema_retrieve
  
def schema_retrieve(question_query):

"""为大模型规划（plan）准备 schema。

  

目标：

- Select 侧：从 params_schema_config 中召回 TopK 个与 query 最相关的属性 schema（用于 prompt 的 <select>/<order>）。

- Where 侧：返回固定的 where-schema（用于 <where>/<group_by>），这里不做召回。

  

返回：

- select_schema: 默认 select schema + TopK 召回项拼接后的字符串

- where_schema: 固定的默认 where schema

- select_retrieve: 仅 TopK 召回项的 schema_select 拼接（便于调试/观测）

- extra_result: 额外信息（TopK 明细、中文字段到 schema 的映射，供下游做字段映射/聚合属性展开）

"""

select_schema = default_shema_select

where_schema = default_shema_where

select_retrieve = ""

extra_result = {}

  

question_query = question_query.lower()

  

# 这些聚合属性在 default_shema_select 里已经覆盖；这里将其从“候选召回池”里排除，避免重复且减少计算量。

default_select_params = [

"car_price_agg",

"car_appearance_agg",

"car_interior_agg",

"car_space_agg",

"car_maneuverability_agg",

"car_power_agg",

"car_endurance_agg",

"car_comfort_agg",

"car_intelligentization_agg",

"car_safety_agg",

"car_energy_consumption_agg",

]

  

# 不参与召回的属性（通常是“兜底/过大集合”的属性）。

unuse_params = ["car_introduction_agg", "car_params_agg"]

  

# select 侧召回数量

topk = 10

  

# {

# "params_cn_name":"销售状态",

# "type":"字面值类型",

# "unit":"",

# "enum":{

# "0":"在售",

# "1":"停售",

# "2":"停产在售",

# "3":"即将上市",

# "4":"订单生产",

# "5":"未上市"

# },

# "data_type":"string",

# "category":"汽车基础信息类属性",

# "nike_names":"",

# "params_exp":"该车的销售状态为在售，销售状态的枚举值包括:在售、停售、停产在售、即将上市、订单生产、未上市。",

# "schema_relevance":"销售状态，示例：该车的销售状态为在售，销售状态的枚举值包括:在售、停售、停产在售、即将上市、订单生产、未上市。属性枚举值包括在售、停售、停产在售、即将上市、订单生产、未上市。",

# "schema_markdown":"|特征字段名称|所属类别|属性示例|枚举值|\n|---|---|---|---|\n|销售状态|汽车基础信息类属性|该车的销售状态为在售，销售状态的枚举值包括:在售、停售、停产在售、即将上市、订单生产、未上市。|在售、停售、停产在售、即将上市、订单生产、未上市|",

# "schema_select":"|销售状态|汽车基础信息类属性|该车的销售状态为在售，销售状态的枚举值包括:在售、停售、停产在售、即将上市、订单生产、未上市。|在售、停售、停产在售、即将上市、订单生产、未上市|",

# "schema_where":""

# }

# -------- 1) 构建候选池 + 中文字段映射（供下游把 LLM 输出的中文字段映射回 params_name/type）--------

params_schema_config_list = list()

params_name_2_schema_info = dict()

for param_name, param_item in params_schema_config.items():

param_cn_name = param_item["params_cn_name"]

params_name_2_schema_info[param_cn_name] = param_item

  

# 跳过默认聚合属性/不参与召回属性，仅对剩余候选做相关性排序。

if param_name in default_select_params or param_name in unuse_params:

continue

  

params_schema_config_list.append(param_item)

  

print("input params_schema_config_list size::: {}".format(len(params_schema_config_list)))

  

# -------- 2) 批量构造 query/title/docs，供相关性模型打分 --------

# query_batch: 同一个 query 对所有候选属性进行打分

query_batch = [question_query] * len(params_schema_config_list)

  

# title_batch: 这里用 params_name 作为 title（给模型一个“候选属性名”的提示）

param_name_batch = [item["params_name"].lower() for item in params_schema_config_list]

  

# docs_batch: 用 schema_relevance（属性描述 + 示例 + 枚举值）作为 docs

schema_relevance_batch = [item["schema_relevance"].lower() for item in params_schema_config_list]

  

# -------- 3) 两路得分：QA/no-answer 相关性分 + BM25 分 --------

# BM25 这里传入的是字符串列表，实际是“字符级 BM25”。

bm25_schema_relevance = BM25(schema_relevance_batch)

  

# 调用 search plugin summary：返回 SPSummaryModelResult，其中 no_answer_score 越低代表越“相关/可回答”。

ans_s_time = time.time()

for _ in range(10):

_, summary_model_results = get_summary_info(query_batch, param_name_batch, schema_relevance_batch)

if len(summary_model_results) > 0:

break

else:

print("get_summary_info failed, try again")

ans_e_time = time.time()

print("ans_cost_time::: {}".format(ans_e_time - ans_s_time))

  

ans_score_list = []

bm25_score_list = []

for idx, ans_result in enumerate(summary_model_results):

# answer_score：1 - no_answer_score，越大越相关

ans_score_list.append(1 - ans_result.no_answer_score)

bm25_score_list.append(bm25_schema_relevance.score(query_batch[idx], idx))

  

# -------- 4) 归一化并融合成 rank_score --------

ans_score_list = min_max_normalize(ans_score_list)

bm25_score_list = min_max_normalize(bm25_score_list)

  

for idx in range(len(params_schema_config_list)):

params_schema_config_list[idx]["answer_score"] = ans_score_list[idx]

params_schema_config_list[idx]["bm25_score"] = bm25_score_list[idx]

params_schema_config_list[idx]["rank_score"] = (

params_schema_config_list[idx]["answer_score"] + params_schema_config_list[idx]["bm25_score"]

)

  

# -------- 5) 排序、取 TopK、拼接 select_schema（写入 prompt）--------

params_schema_config_list = sorted(params_schema_config_list, key=lambda x: x["rank_score"], reverse=True)

  

# 调试输出：保存全量候选的打分排序结果

dump_json(params_schema_config_list, "/cloudide/workspace/motor_dynamic_summary/eval_data/params_schema_config_list.json")

  

top_schema_config_list = params_schema_config_list[:topk]

select_retrieve = "\n".join([item["schema_select"] for item in top_schema_config_list])

select_schema += "\n" + select_retrieve

  

extra_result = {

"top_schema_config_list": top_schema_config_list,

"params_name_2_schema_info": params_name_2_schema_info,

}

  

return select_schema, where_schema, select_retrieve, extra_result
